-- 3. Tabla de configuración global (para T&C editables)
CREATE TABLE IF NOT EXISTS configuracion (
  clave  text PRIMARY KEY,
  valor  text NOT NULL,
  actualizado_en timestamptz DEFAULT now()
);

-- 4. Tabla de sub-items por categoría
CREATE TABLE IF NOT EXISTS categoria_subitems (

  nombre          text NOT NULL,
  precio          numeric,
  porcentaje_tasa numeric NOT NULL DEFAULT 0,
  activo          boolean NOT NULL DEFAULT true,

### Ítem 3 — Técnico acepta el trabajo
**Nuevo estado `asignada`** en el flujo:
- Admin asigna técnico → estado pasa a `asignada` (antes iba directo a `aceptada`)

- Si rechaza → estado = `pendiente`, `tecnico_id = null` (desasignado)

- `src/lib/types.ts`: `SolicitudEstado` incluye `'asignada'`
