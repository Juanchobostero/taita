**BD:**
```sql
CREATE OR REPLACE FUNCTION recalcular_calificacion(p_tecnico_id UUID)
RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  UPDATE tecnicos SET
    calificacion_promedio = (
      SELECT COALESCE(ROUND(AVG(calificacion)::numeric, 1), 0)
      FROM resenas WHERE tecnico_id = p_tecnico_id
    ),
    total_servicios = (
      SELECT COUNT(*) FROM resenas WHERE tecnico_id = p_tecnico_id
    )
  WHERE id = p_tecnico_id;
END;
$$;
```
